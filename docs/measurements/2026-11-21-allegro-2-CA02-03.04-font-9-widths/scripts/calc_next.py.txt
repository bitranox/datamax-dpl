from gen_next import SETS, RIGHT, LEFT_LIMIT, line_text
from gen_calib import PROFILE, DOTS_PER_MM
from datamax_dpl.domain.label import Text
from datamax_dpl.domain.text_metrics import char_advances, font9_em_dots
read = {int(l.split()[0]): l.split()[1] for l in open("next/readings.txt")}
mm = lambda v: (1 if v[0] == "S" else -1) * float(v[1:])
n = 42
for points, _y, _p, _r, _per, target, chars in SETS:
    em = font9_em_dots(points, PROFILE)
    m_text, m_n, _ = line_text("M", points, min(target, RIGHT - LEFT_LIMIT))
    m_err = mm(read[n]) / (m_n + 1)
    for c in chars:
        text, cnt, _ = line_text(c, points, min(target, RIGHT - LEFT_LIMIT))
        pred = char_advances(Text(x=0, y=0, text=c, font=9, points=points), PROFILE)[0]
        err = m_err if c == "M" else (mm(read[n]) - m_err) / cnt
        real = pred - err * DOTS_PER_MM
        print(f"{n} {points:2d}pt {c!r:5} N={cnt:2d} {read[n]:6} pred {pred:3d} real {real:6.2f} dots = {real/em*1000:6.1f}/1000 em (em {em:.2f})")
        n += 1
